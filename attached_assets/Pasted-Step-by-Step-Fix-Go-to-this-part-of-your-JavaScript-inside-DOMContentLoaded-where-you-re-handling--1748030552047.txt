<script>
// Customer Selection Modal Logic
document.addEventListener('DOMContentLoaded', function () {
    const searchBtn = document.getElementById('searchCustomerBtn');
    const searchInput = document.getElementById('customerSearchInput');
    const customerRows = document.querySelectorAll('#customerTable tbody tr');
    const noResultsMessage = document.getElementById('noCustomersFound');

    // Search Function
    function searchCustomers() {
        const searchTerm = searchInput.value.toLowerCase().trim();
        let resultsFound = false;

        customerRows.forEach(row => {
            const name = row.cells[0].textContent.toLowerCase();
            const email = row.cells[1].textContent.toLowerCase();
            const id = row.cells[2].textContent.toLowerCase();

            const isMatch = name.includes(searchTerm) || email.includes(searchTerm) || id.includes(searchTerm);
            row.style.display = isMatch ? '' : 'none';
            if (isMatch) resultsFound = true;
        });

        if (noResultsMessage) {
            noResultsMessage.classList.toggle('d-none', resultsFound);
        }
    }

    if (searchBtn && searchInput) {
        searchBtn.addEventListener('click', searchCustomers);
        searchInput.addEventListener('keypress', function (e) {
            if (e.key === 'Enter') {
                searchCustomers();
                e.preventDefault();
            }
        });
    }

    // Handle Customer Selection
    const selectButtons = document.querySelectorAll('.customer-select-btn');
    selectButtons.forEach(button => {
        button.addEventListener('click', () => {
            const customerId = button.getAttribute('data-id');
            const customerName = button.getAttribute('data-name');
            const customerEmail = button.getAttribute('data-email');

            // Create form to submit customer selection
            const form = document.createElement('form');
            form.method = 'POST';
            form.action = '/booking/create-from-detail';
            form.style.display = 'none';

            const inputs = [
                { name: 'customer', value: customerId },
                { name: 'request_id', value: generateRequestId() }
            ];

            const csrfTokenMeta = document.querySelector('meta[name="csrf-token"]');
            if (csrfTokenMeta) {
                inputs.push({ name: 'csrf_token', value: csrfTokenMeta.getAttribute('content') });
            }

            inputs.forEach(({ name, value }) => {
                const input = document.createElement('input');
                input.type = 'hidden';
                input.name = name;
                input.value = value;
                form.appendChild(input);
            });

            document.body.appendChild(form);

            // Update UI fields immediately (optional)
            updateCustomerUI(customerName, customerEmail, customerId);

            // Close modal safely
            const modalEl = document.getElementById('customerSelectionModal');
            if (modalEl) {
                const modal = bootstrap.Modal.getOrCreateInstance(modalEl);
                modal.hide();
            }

            // Feedback & submit
            alert(`Customer ${customerName} selected successfully! Creating booking...`);
            form.submit();
        });
    });

    // Helper: Request ID generator
    function generateRequestId() {
        const timestamp = Date.now().toString(36).slice(-6);
        const randomPart = Math.random().toString(36).substring(2, 5).toUpperCase();
        return `IR-${timestamp}${randomPart}`;
    }

    // Helper: Update UI with selected customer
    function updateCustomerUI(name, email, id) {
        const nameEl = document.getElementById('customerName');
        const emailEl = document.getElementById('customerEmail');
        const idEl = document.getElementById('customerId');

        if (nameEl) nameEl.textContent = name;
        if (emailEl) emailEl.textContent = email;
        if (idEl) idEl.textContent = id;
    }
});
</script>
{% endblock %}
