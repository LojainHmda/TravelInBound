        function getStatusBadge(status) {
            const badges = {
                'pending': 'bg-yellow-100 text-yellow-800 dark:bg-yellow-900 dark:text-yellow-300',
                'confirmed': 'bg-blue-100 text-blue-800 dark:bg-blue-900 dark:text-blue-300',
                'completed': 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-300',
                'cancelled': 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-300'
            };
            return `<span class="px-2 py-1 text-xs font-medium rounded-full ${badges[status] || 'bg-gray-100 text-gray-800 dark:bg-gray-700 dark:text-gray-300'}">${status.charAt(0).toUpperCase() + status.slice(1)}</span>`;
        }

        function renderBookings(bookings) {
            const tbody = document.getElementById('bookingsTable');
            tbody.innerHTML = bookings.map(booking => `
                <tr class="hover:bg-gray-50 dark:hover:bg-gray-700 transition-colors">
                    <td class="px-6 py-4 whitespace-nowrap">
                        <div class="text-sm font-medium text-gray-900 dark:text-white">${booking.reference}</div>
                    </td>
                    <td class="px-6 py-4 whitespace-nowrap">
                        <div class="text-sm font-medium text-gray-900 dark:text-white">${booking.customer}</div>
                        <div class="text-sm text-gray-500 dark:text-gray-400">${booking.email}</div>
                    </td>
                    <td class="px-6 py-4 whitespace-nowrap text-sm text-gray-900 dark:text-white">
                        ${new Date(booking.created).toLocaleDateString('en-US', { 
                            year: 'numeric', 
                            month: 'short', 
                            day: 'numeric' 
                        })}
                    </td>
                    <td class="px-6 py-4 whitespace-nowrap">
                        ${getStatusBadge(booking.status)}
                    </td>
                    <td class="px-6 py-4 whitespace-nowrap text-sm text-gray-900 dark:text-white">
                        ${booking.services} service${booking.services !== 1 ? 's' : ''}

                        ${booking.totalAmount}

                        <button onclick="viewBooking('${booking.reference}')"

                        <button onclick="editBooking('${booking.reference}')" 
                                class="text-booking-yellow hover:text-yellow-600 mr-3 transition-colors">Edit</button>
                        <button onclick="deleteBooking('${booking.reference}')" 
                                class="text-red-600 hover:text-red-700 transition-colors">Delete</button>
                    </td>
                </tr>
            `).join('');

            document.getElementById('resultsCount').textContent = `Showing ${bookings.length} of ${sampleBookings.length} bookings`;
        }

        function searchBookings() {
            const reference = document.getElementById('referenceSearch').value.toLowerCase();
            const customer = document.getElementById('customerSearch').value.toLowerCase();
            const status = document.getElementById('statusFilter').value;
            const dateFrom = document.getElementById('dateFrom').value;

            currentBookings = sampleBookings.filter(booking => {
                const matchesReference = !reference || booking.reference.toLowerCase().includes(reference);
                const matchesCustomer = !customer || booking.customer.toLowerCase().includes(customer);
                const matchesStatus = !status || booking.status === status;
                const matchesDate = !dateFrom || booking.created >= dateFrom;

                return matchesReference && matchesCustomer && matchesStatus && matchesDate;
            });

            renderBookings(currentBookings);
        }

        function clearFilters() {
            document.getElementById('referenceSearch').value = '';
            document.getElementById('customerSearch').value = '';
            document.getElementById('statusFilter').value = '';
            document.getElementById('dateFrom').value = '';
            currentBookings = [...sampleBookings];
            renderBookings(currentBookings);
        }

        function sortResults() {
            const sortBy = document.getElementById('sortBy').value;
            
            currentBookings.sort((a, b) => {
                switch(sortBy) {
                    case 'date-desc':
                        return new Date(b.created) - new Date(a.created);
                    case 'date-asc':
                        return new Date(a.created) - new Date(b.created);
                    case 'reference':
                        return a.reference.localeCompare(b.reference);
                    case 'customer':
                        return a.customer.localeCompare(b.customer);
                    case 'status':
                        return a.status.localeCompare(b.status);
                    default:
                        return 0;
                }
            });

            renderBookings(currentBookings);
        }

            alert(`Viewing booking: ${reference}`);

        }

        function editBooking(reference) {
            alert(`Editing booking: ${reference}`);
            // In a real app, this would open the booking edit interface
        }

        function deleteBooking(reference) {
            if (confirm(`Are you sure you want to delete booking ${reference}?`)) {
                currentBookings = currentBookings.filter(b => b.reference !== reference);
                renderBookings(currentBookings);
            }
        }
