        function showMessage(text, type = 'info') {
            const messageBox = document.getElementById('customMessageBox');
            const messageText = document.getElementById('messageText');
            
            messageText.textContent = text;
            messageBox.className = 'message-box'; // Reset classes
            if (type === 'success') {
                messageBox.classList.add('message-box-success');
            } else if (type === 'error') {

            } else { // default to info

            }
            messageBox.style.display = 'block';

        }

            }

        // --- Booking Data Handling and Rendering ---
        function getStatusBadge(status) {
            const badges = {
                'pending': 'bg-yellow-100 text-yellow-800 dark:bg-yellow-900 dark:text-yellow-300',
                'confirmed': 'bg-blue-100 text-blue-800 dark:bg-blue-900 dark:text-blue-300',
                'completed': 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-300',
                'cancelled': 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-300'
            };
            // Capitalize first letter of status
            const capitalizedStatus = status.charAt(0).toUpperCase() + status.slice(1);
            return `<span class="px-2 py-1 text-xs font-medium rounded-full ${badges[status.toLowerCase()] || 'bg-gray-100 text-gray-800 dark:bg-gray-700 dark:text-gray-300'}">${capitalizedStatus}</span>`;
        }

        // --- Search, Filter, Sort Functions ---
        function searchBookings() {
            // Get filter values
            const reference = document.getElementById('referenceSearch').value.toLowerCase().trim();
            const customer = document.getElementById('customerSearch').value.toLowerCase().trim();
            const status = document.getElementById('statusFilter').value.toLowerCase(); // Ensure lowercase for comparison
            const dateFrom = document.getElementById('dateFrom').value;

            // Filter the original sampleBookings array
            currentBookings = sampleBookings.filter(booking => {
                const matchesReference = !reference || booking.reference.toLowerCase().includes(reference);
                const matchesCustomer = !customer || booking.customer.toLowerCase().includes(customer);
                const matchesStatus = !status || booking.status.toLowerCase() === status;
                // Date comparison: only if dateFrom is set, and booking.created is on or after dateFrom
                const matchesDate = !dateFrom || (booking.created && new Date(booking.created) >= new Date(dateFrom));
                
                return matchesReference && matchesCustomer && matchesStatus && matchesDate;
            });
            
            sortResults(); // Apply current sort order to filtered results
            // renderBookings(currentBookings); // renderBookings is called within sortResults
            if (currentBookings.length > 0) {
                showMessage(`${currentBookings.length} booking(s) found.`, 'success');
            } else {
                showMessage('No bookings match your criteria.', 'info');
            }
        }

        function clearFilters() {
            // Reset input fields
            document.getElementById('referenceSearch').value = '';
            document.getElementById('customerSearch').value = '';
            document.getElementById('statusFilter').value = '';
            document.getElementById('dateFrom').value = '';
            
            // Reset currentBookings to the full list and re-render
            currentBookings = [...sampleBookings];
            sortResults(); // Apply default sort order
            // renderBookings(currentBookings); // renderBookings is called within sortResults
            showMessage('Filters cleared. Showing all bookings.', 'info');
        }

        function sortResults() {
            const sortBy = document.getElementById('sortBy').value;
            
            currentBookings.sort((a, b) => {
                switch(sortBy) {
                    case 'date-desc':
                        return new Date(b.created) - new Date(a.created);
                    case 'date-asc':
                        return new Date(a.created) - new Date(b.created);
                    case 'reference':
                        return a.reference.localeCompare(b.reference);
                    case 'customer':
                        return a.customer.localeCompare(b.customer);
                    case 'status':
                        return a.status.localeCompare(b.status);
                    default:
                        return 0;
                }
            });
            renderBookings(currentBookings); // Re-render after sorting
        }

        function triggerDeleteBooking(reference) {

        }

        function exportResults() {
            // In a real app, this would generate and download a CSV/Excel file
            if (currentBookings.length === 0) {
                showMessage('No data to export.', 'info');
                return;
            }
            // Basic CSV generation example (can be expanded)
            let csvContent = "data:text/csv;charset=utf-8,";
            // Headers
            csvContent += "Reference,Customer,Email,Created Date,Status,Services,Total Amount\n";
            // Rows
            currentBookings.forEach(booking => {
                let row = [
                    booking.reference,
                    booking.customer,
                    booking.email,
                    booking.created,
                    booking.status,
                    booking.services,
                    `"${booking.totalAmount.replace('$', '')}"` // Handle potential commas in amount
                ].join(",");
                csvContent += row + "\r\n";
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "booking_results.csv");
            document.body.appendChild(link); // Required for Firefox
            link.click();
            document.body.removeChild(link);
            showMessage('Results exported as CSV.', 'success');
        }
