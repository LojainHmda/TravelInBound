// Function to close the summary modal
function closeSummaryModal() {
    const modal = document.getElementById('summaryModal');
    if (modal) {
        modal.remove();
    }
}

// Function to print the summary table
function printSummaryTable() {
    const modalContent = document.querySelector('#summaryModal .bg-white');
    if (modalContent) {
        const printWindow = window.open('', '_blank');
        printWindow.document.write(`
            <html>
                <head>
                    <title>Trip Summary - {{ request.request_number }}</title>
                    <style>
                        body { font-family: Arial, sans-serif; margin: 20px; }
                        table { width: 100%; border-collapse: collapse; margin-bottom: 20px; }
                        th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
                        th { background-color: #f2f2f2; font-weight: bold; }
                        h2, h3 { color: #333; }
                        @media print {
                            .no-print { display: none; }
                        }
                    </style>
                </head>
                <body>
                    ${modalContent.innerHTML}
                </body>
            </html>
        `);
        printWindow.document.close();
        printWindow.print();
    }
}

// Function to export summary to Excel
function exportSummaryToExcel() {
    const wb = XLSX.utils.book_new();
    
    // Collect all data
    const summaryData = [];
    
    // Header
    summaryData.push({
        'Section': 'Trip Information',
        'Request Number': '{{ request.request_number }}',
        'Customer': document.getElementById('selectedCustomerName')?.value || 'N/A',
        'Dates': `${document.getElementById('fromDate')?.value || ''} - ${document.getElementById('toDate')?.value || ''}`,
        'PAX': document.querySelector('[name="pax"]')?.value || '0',
        'Days': document.getElementById('noOfDays')?.value || '0'
    });
    
    summaryData.push({});
    
    // Hotels
    const hotelRows = document.querySelectorAll('#hotels-table-body tr');
    if (hotelRows.length > 0) {
        summaryData.push({
            'Section': 'Hotels',
            'Hotel Name': 'Hotel Name',
            'Check-in': 'Check-in',
            'Check-out': 'Check-out',
            'Nights': 'Nights',
            'Status': 'Status'
        });
        
        hotelRows.forEach(row => {
            const cells = row.cells;
            const dates = cells[1].textContent.trim().split(' - ');
            summaryData.push({
                'Section': '',
                'Hotel Name': cells[0].textContent.trim(),
                'Check-in': dates[0] || '',
                'Check-out': dates[1] || '',
                'Nights': cells[2].textContent.trim(),
                'Status': cells[4].textContent.trim()
            });
        });
    }
    
    summaryData.push({});
    
    // Flights
    const arrivalRows = document.querySelectorAll('#arrival-table-body tr');
    const departureRows = document.querySelectorAll('#departure-table-body tr');
    
    if (arrivalRows.length > 0 || departureRows.length > 0) {
        summaryData.push({
            'Section': 'Flights',
            'Type': 'Type',
            'Date': 'Date',
            'Time': 'Time',
            'Point': 'Point',
            'Flight No.': 'Flight No.',
            'PAX': 'PAX'
        });
        
        arrivalRows.forEach(row => {
            const cells = row.cells;
            summaryData.push({
                'Section': '',
                'Type': 'Arrival',
                'Date': cells[0].textContent.trim(),
                'Time': cells[1].textContent.trim(),
                'Point': cells[2].textContent.trim(),
                'Flight No.': cells[3].textContent.trim(),
                'PAX': cells[4].textContent.trim()
            });
        });
        
        departureRows.forEach(row => {
            const cells = row.cells;
            summaryData.push({
                'Section': '',
                'Type': 'Departure',
                'Date': cells[0].textContent.trim(),
                'Time': cells[1].textContent.trim(),
                'Point': cells[2].textContent.trim(),
                'Flight No.': cells[3].textContent.trim(),
                'PAX': cells[4].textContent.trim()
            });
        });
    }
    
    // Create worksheet
    const ws = XLSX.utils.json_to_sheet(summaryData);
    
    // Set column widths
    const wscols = [
        {wch: 15}, // Section
        {wch: 25}, // Request Number / Hotel Name / Type
        {wch: 20}, // Customer / Check-in / Date
        {wch: 20}, // Dates / Check-out / Time
        {wch: 15}, // PAX / Nights / Point
        {wch: 15}, // Days / Status / Flight No.
        {wch: 10}  // Extra column
    ];
    ws['!cols'] = wscols;
    
    XLSX.utils.book_append_sheet(wb, ws, "Trip Summary");
    
    // Save file
    XLSX.writeFile(wb, `Trip_Summary_{{ request.request_number }}_${new Date().toISOString().split('T')[0]}.xlsx`);
}

// Add button to trigger the summary table
function addSummaryButton() {
    const tripSummaryCard = document.getElementById('tripSummaryCard');
    if (tripSummaryCard) {
        const header = tripSummaryCard.querySelector('.px-3.py-2.border-b');
        if (header) {
            const button = document.createElement('button');
            button.className = 'px-3 py-1.5 bg-purple-600 text-white rounded-lg hover:bg-purple-700 transition text-xs font-medium flex items-center gap-2';
            button.innerHTML = '<i class="fas fa-table"></i> View Full Summary';
            button.onclick = generateSummaryTable;
            header.appendChild(button);
        }
    }
}

// Initialize on page load
document.addEventListener('DOMContentLoaded', function() {
    addSummaryButton();
});
