        function getStatusBadge(status) {
            const badges = {
                'request': 'bg-gray-100 text-gray-800 dark:bg-gray-700 dark:text-gray-300',
                'in-progress': 'bg-travel-yellow text-gray-900',
                'confirmed': 'bg-green-100 text-green-800 dark:bg-green-900 dark:text-green-300',
                'cancelled': 'bg-red-100 text-red-800 dark:bg-red-900 dark:text-red-300'
            };
            const icons = {
                'request': 'fas fa-file-alt',
                'in-progress': 'fas fa-cog fa-spin',
                'confirmed': 'fas fa-check',
                'cancelled': 'fas fa-times'
            };
            return `<span class="inline-flex items-center px-3 py-1 text-xs font-medium rounded-full ${badges[status] || 'bg-gray-100 text-gray-800 dark:bg-gray-700 dark:text-gray-300'}">
                <i class="${icons[status] || 'fas fa-question'} mr-1"></i>
                ${status === 'in-progress' ? 'In Progress' : status.charAt(0).toUpperCase() + status.slice(1)}
            </span>`;
        }

        function searchBookings() {
            const reference = document.getElementById('referenceSearch').value.toLowerCase();
            const customer = document.getElementById('customerSearch').value.toLowerCase();
            const status = document.getElementById('statusFilter').value;
            const dateFrom = document.getElementById('dateFrom').value;

            currentBookings = sampleBookings.filter(booking => {
                const matchesReference = !reference || booking.reference.toLowerCase().includes(reference);
                const matchesCustomer = !customer || booking.customer.toLowerCase().includes(customer);
                const matchesStatus = !status || booking.status === status;
                const matchesDate = !dateFrom || booking.created >= dateFrom;

                return matchesReference && matchesCustomer && matchesStatus && matchesDate;
            });

            renderBookings(currentBookings);
        }

        function clearFilters() {
            document.getElementById('referenceSearch').value = '';
            document.getElementById('customerSearch').value = '';
            document.getElementById('statusFilter').value = '';
            document.getElementById('dateFrom').value = '';
            currentBookings = [...sampleBookings];
            renderBookings(currentBookings);
        }

        function sortResults() {
            const sortBy = document.getElementById('sortBy').value;
            
            currentBookings.sort((a, b) => {
                switch(sortBy) {
                    case 'date-desc':
                        return new Date(b.created) - new Date(a.created);
                    case 'date-asc':
                        return new Date(a.created) - new Date(b.created);
                    case 'reference':
                        return a.reference.localeCompare(b.reference);
                    case 'customer':
                        return a.customer.localeCompare(b.customer);
                    case 'status':
                        return a.status.localeCompare(b.status);
                    default:
                        return 0;
                }
            });

            renderBookings(currentBookings);
        }

                currentBookings = currentBookings.filter(b => b.reference !== reference);
                renderBookings(currentBookings);
            }
        }
