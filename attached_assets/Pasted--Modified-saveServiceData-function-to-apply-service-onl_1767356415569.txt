// Modified saveServiceData function to apply service only to specific date range
function saveServiceData(serviceType) {
    // Validate date range for global services
    if (['hotel', 'guide', 'transport', 'meal'].includes(serviceType)) {
        const dateValidation = validateDateRange(serviceType);
        if (!dateValidation.valid) {
            showInlineError(serviceType, dateValidation.message);
            return;
        }
    }
    
    // Validate room distribution for hotels
    if (serviceType === 'hotel') {
        const validation = validateRoomDistribution();
        if (!validation.valid) {
            showInlineError('hotel', validation.message);
            return;
        }
    }
    
    // Collect form data
    const activePanel = document.getElementById(`service-panel-${serviceType}`);
    const formData = {};
    
    const fields = activePanel 
        ? activePanel.querySelectorAll(`[data-service="${serviceType}"]`)
        : document.querySelectorAll(`[data-service="${serviceType}"]`);
    
    fields.forEach(field => {
        if (field.id) {
            formData[field.id] = field.value;
        }
    });
    
    // Determine date range for application
    let fromDate, toDate;
    
    switch(serviceType) {
        case 'hotel':
            fromDate = document.getElementById('hotel_check_in')?.value;
            toDate = document.getElementById('hotel_check_out')?.value;
            break;
        case 'guide':
            fromDate = document.getElementById('guide_from_date')?.value;
            toDate = document.getElementById('guide_to_date')?.value;
            break;
        case 'transport':
            fromDate = document.getElementById('transport_from_date')?.value;
            toDate = document.getElementById('transport_to_date')?.value;
            break;
        case 'meal':
            fromDate = document.getElementById('meal_from_date')?.value;
            toDate = document.getElementById('meal_to_date')?.value;
            break;
        case 'arrival':
            fromDate = document.getElementById('arrival_date')?.value;
            toDate = document.getElementById('arrival_date')?.value;
            break;
        case 'departure':
            fromDate = document.getElementById('departure_date')?.value;
            toDate = document.getElementById('departure_date')?.value;
            break;
    }
    
    const payload = {
        service_type: serviceType,
        data: formData,
        // Apply only to specific date range
        apply_to_specific_days: true,
        from_date: fromDate,
        to_date: toDate
    };

    // Add context information
    if (currentServiceContext.rowId) {
        payload.row_id = currentServiceContext.rowId;
        payload.is_global = false;
    } else {
        payload.is_global = true;
        // For global services, apply only to specified date range
        payload.apply_to_specific_days = true;
    }
    
    fetch(`/inbound/api/${requestId}/save-service-data`, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': csrfToken
        },
        body: JSON.stringify(payload)
    })
    .then(response => response.json().then(data => ({ ok: response.ok, data })))
    .then(({ ok, data }) => {
        if (ok && data.success) {
            clearInlineError(serviceType);
            showNotification(data.message || 'Service data saved!', 'success');
            
            // Update itinerary if needed
            if (data.itinerary_html) {
                const tbody = document.querySelector('#itineraryTable tbody');
                if (tbody) tbody.innerHTML = data.itinerary_html;
            }
            
            // Update service table
            updateServiceTable(serviceType, data);
            
            // Update summary counts
            updateSummaryCounts();
            
            // NEW: Activate icons only for the specific date range
            if (!currentServiceContext.rowId && fromDate && toDate) {
                activateIconsForDateRange(serviceType, fromDate, toDate);
                const startDate = new Date(fromDate).toLocaleDateString('en-US', { day: 'numeric', month: 'short' });
                const endDate = new Date(toDate).toLocaleDateString('en-US', { day: 'numeric', month: 'short' });
                showNotification(`Service applied to days ${startDate} - ${endDate} only`, 'info');
            } else if (currentServiceContext.rowId) {
                // For day-specific services, activate only that day's icon
                activateIcon(currentServiceContext.rowId, serviceType);
            }
        } else {
            const errorMsg = data.error || 'Error saving service data';
            showInlineError(serviceType, errorMsg);
        }
    })
    .catch(error => {
        console.error('Error:', error);
        showNotification('Error saving service data', 'error');
    });
    
    // Clear form fields after saving for easy addition of another item
    setTimeout(() => {
        if (!currentServiceContext.rowId) {
            clearServiceForm(serviceType);
            showNotification('Saved. You can add another item.', 'success');
        }
    }, 500);
}

// NEW: Function to activate icons only for rows within a specific date range
function activateIconsForDateRange(serviceType, fromDate, toDate) {
    const serviceIconMap = {
        'hotel': 'hotel',
        'guide': 'guide',
        'transport': 'transport',
        'meal': 'meal',
        'arrival': 'arrival',
        'departure': 'departure'
    };
    
    const iconClass = serviceIconMap[serviceType];
    if (!iconClass) return;
    
    // Convert dates to Date objects for comparison
    const startDate = new Date(fromDate);
    const endDate = new Date(toDate);
    
    // Find all rows in the itinerary
    const rows = document.querySelectorAll('#itineraryTableBody tr[data-row-id]');
    
    rows.forEach(row => {
        // Get the date for this row (assuming it's stored in data-date attribute or in a specific cell)
        let rowDateStr = row.dataset.date;
        if (!rowDateStr) {
            // Try to get date from the date cell (second column)
            const dateCell = row.cells[1];
            if (dateCell) {
                rowDateStr = dateCell.textContent.trim();
            }
        }
        
        if (rowDateStr) {
            // Parse the row date (format might be "DD Mon" or "DD Mon YYYY")
            const rowDate = parseDateFromText(rowDateStr);
            
            // Check if this row's date is within the specified range
            if (rowDate && rowDate >= startDate && rowDate <= endDate) {
                // Activate the icon for this service type on this row
                const icon = row.querySelector(`.service-icon-toggle.${iconClass}`);
                if (icon) {
                    icon.classList.add('active');
                }
            }
        }
    });
}

// Helper function to parse date from text (handles various formats)
function parseDateFromText(dateText) {
    // Handle formats like "15 Jun" or "15 Jun 2023"
    const parts = dateText.trim().split(' ');
    if (parts.length < 2) return null;
    
    const day = parseInt(parts[0]);
    const monthStr = parts[1];
    
    // Convert month name to number (0-indexed)
    const months = {
        'Jan': 0, 'Feb': 1, 'Mar': 2, 'Apr': 3, 'May': 4, 'Jun': 5,
        'Jul': 6, 'Aug': 7, 'Sep': 8, 'Oct': 9, 'Nov': 10, 'Dec': 11
    };
    
    const month = months[monthStr];
    if (month === undefined) return null;
    
    // Use current year if not specified
    const year = parts.length > 2 ? parseInt(parts[2]) : new Date().getFullYear();
    
    return new Date(year, month, day);
}

// Modified updateServiceContextUI to show application scope
function updateServiceContextUI(serviceName, rowId) {
    const badge = document.getElementById('panel-context-badge-' + serviceName);
    if (!badge) return;

    if (rowId) {
        badge.textContent = `Day Specific (ID: ${rowId})`;
        badge.className = "text-xs font-semibold px-2 py-0.5 rounded bg-blue-100 text-blue-700";
        populateFormWithDayData(serviceName, rowId);
    } else {
        // For global services, show the specific date range
        let fromDate, toDate;
        
        switch(serviceName) {
            case 'hotel':
                fromDate = document.getElementById('hotel_check_in')?.value;
                toDate = document.getElementById('hotel_check_out')?.value;
                break;
            case 'guide':
                fromDate = document.getElementById('guide_from_date')?.value;
                toDate = document.getElementById('guide_to_date')?.value;
                break;
            case 'transport':
                fromDate = document.getElementById('transport_from_date')?.value;
                toDate = document.getElementById('transport_to_date')?.value;
                break;
            case 'meal':
                fromDate = document.getElementById('meal_from_date')?.value;
                toDate = document.getElementById('meal_to_date')?.value;
                break;
            case 'arrival':
                fromDate = document.getElementById('arrival_date')?.value;
                toDate = document.getElementById('arrival_date')?.value;
                break;
            case 'departure':
                fromDate = document.getElementById('departure_date')?.value;
                toDate = document.getElementById('departure_date')?.value;
                break;
        }
        
        if (fromDate && toDate) {
            const startDate = new Date(fromDate).toLocaleDateString('en-US', { day: 'numeric', month: 'short' });
            const endDate = new Date(toDate).toLocaleDateString('en-US', { day: 'numeric', month: 'short' });
            badge.textContent = `Specific Days (${startDate} - ${endDate})`;
            badge.className = "text-xs font-semibold px-2 py-0.5 rounded bg-amber-100 text-amber-700";
        } else {
            badge.textContent = "Global (Applies to All)";
            badge.className = "text-xs font-normal px-2 py-0.5 rounded bg-slate-100 text-slate-500";
        }
        
        clearServiceForm(serviceName);
    }
    
    if (serviceName === 'hotel') {
        initializeRoomDistribution();
    }
}

// Add event listeners to update context badge when dates change
document.addEventListener('DOMContentLoaded', function() {
    // For services with dates, add event listeners to update badge
    ['hotel', 'guide', 'transport', 'meal', 'arrival', 'departure'].forEach(serviceType => {
        const dateFields = {
            'hotel': ['hotel_check_in', 'hotel_check_out'],
            'guide': ['guide_from_date', 'guide_to_date'],
            'transport': ['transport_from_date', 'transport_to_date'],
            'meal': ['meal_from_date', 'meal_to_date'],
            'arrival': ['arrival_date'],
            'departure': ['departure_date']
        };
        
        dateFields[serviceType].forEach(fieldId => {
            const field = document.getElementById(fieldId);
            if (field) {
                field.addEventListener('change', function() {
                    // Update badge only if current service matches
                    if (currentServiceContext.serviceName === serviceType && !currentServiceContext.rowId) {
                        updateServiceContextUI(serviceType, null);
                    }
                });
            }
        });
    });
    
    // Rest of initialization code...
});
