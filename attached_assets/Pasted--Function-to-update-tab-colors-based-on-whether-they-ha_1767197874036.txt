// Function to update tab colors based on whether they have data
function updateTabColors() {
    const serviceTabs = {
        'arrival': document.querySelector('.service-btn[data-service="arrival"]'),
        'departure': document.querySelector('.service-btn[data-service="departure"]'),
        'hotel': document.querySelector('.service-btn[data-service="hotel"]'),
        'guide': document.querySelector('.service-btn[data-service="guide"]'),
        'transport': document.querySelector('.service-btn[data-service="transport"]'),
        'meal': document.querySelector('.service-btn[data-service="meal"]')
    };

    // Check each service type for data
    const hasData = {
        'arrival': document.querySelectorAll('#arrival-table-body tr').length > 0,
        'departure': document.querySelectorAll('#departure-table-body tr').length > 0,
        'hotel': document.querySelectorAll('#hotels-table-body tr').length > 0,
        'guide': document.querySelectorAll('#guides-table-body tr').length > 0,
        'transport': document.querySelectorAll('#transports-table-body tr').length > 0,
        'meal': document.querySelectorAll('#meals-table-body tr').length > 0
    };

    // Update tab colors based on data
    Object.keys(serviceTabs).forEach(service => {
        const tab = serviceTabs[service];
        if (tab) {
            const icon = tab.querySelector('i');
            
            if (hasData[service]) {
                // Tab has data - apply filled style
                tab.classList.remove('bg-white', 'text-slate-500');
                tab.classList.add('filled-tab');
                
                // Different colors for different services
                const colorMap = {
                    'arrival': 'bg-green-100 text-green-700 border-green-300',
                    'departure': 'bg-orange-100 text-orange-700 border-orange-300',
                    'hotel': 'bg-blue-100 text-blue-700 border-blue-300',
                    'guide': 'bg-teal-100 text-teal-700 border-teal-300',
                    'transport': 'bg-cyan-100 text-cyan-700 border-cyan-300',
                    'meal': 'bg-amber-100 text-amber-700 border-amber-300'
                };
                
                tab.className = `service-btn inline-flex items-center gap-1 px-3 py-1.5 rounded-full font-medium transition-all ${colorMap[service]}`;
                
                // Update icon color
                if (icon) {
                    icon.classList.remove('text-slate-400');
                    const iconColorMap = {
                        'arrival': 'text-green-600',
                        'departure': 'text-orange-600',
                        'hotel': 'text-blue-600',
                        'guide': 'text-teal-600',
                        'transport': 'text-cyan-600',
                        'meal': 'text-amber-600'
                    };
                    icon.classList.add(iconColorMap[service]);
                }
                
                // Add a small indicator badge
                let badge = tab.querySelector('.data-badge');
                if (!badge) {
                    badge = document.createElement('span');
                    badge.className = 'data-badge ml-1 w-2 h-2 rounded-full';
                    tab.appendChild(badge);
                }
                badge.className = `data-badge ml-1 w-2 h-2 rounded-full ${
                    service === 'arrival' ? 'bg-green-500' :
                    service === 'departure' ? 'bg-orange-500' :
                    service === 'hotel' ? 'bg-blue-500' :
                    service === 'guide' ? 'bg-teal-500' :
                    service === 'transport' ? 'bg-cyan-500' :
                    'bg-amber-500'
                }`;
            } else {
                // Tab has no data - reset to default style
                tab.classList.remove('filled-tab');
                tab.className = 'service-btn inline-flex items-center gap-1 px-3 py-1.5 rounded-full bg-white text-slate-500 text-xs font-medium transition-all';
                
                const icon = tab.querySelector('i');
                if (icon) {
                    icon.className = icon.className.replace(/text-\w+-600/g, 'text-slate-400');
                }
                
                // Remove badge if exists
                const badge = tab.querySelector('.data-badge');
                if (badge) {
                    badge.remove();
                }
            }
        }
    });
}
