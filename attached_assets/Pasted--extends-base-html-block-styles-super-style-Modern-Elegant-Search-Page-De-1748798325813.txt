<script>
function clearFilters() {
    document.getElementById('searchForm').reset();
    updateFilterCount();
    window.location.href = '{{ url_for("booking.search") }}';
}

function exportResults() {
    const params = new URLSearchParams(window.location.search);
    params.set('export', 'csv');
    window.location.href = '{{ url_for("booking.search") }}?' + params.toString();
}

function updateFilterCount() {
    const form = document.getElementById('searchForm');
    const inputs = form.querySelectorAll('input[type="text"], input[type="date"], input[type="number"], select');
    let activeCount = 0;
    
    inputs.forEach(input => {
        if (input.value && input.value.trim() !== '') {
            activeCount++;
        }
    });
    
    const counterElement = document.getElementById('activeFiltersCount');
    if (activeCount === 0) {
        counterElement.textContent = '0 Active Filters';
        counterElement.className = 'badge-counter';
    } else {
        counterElement.textContent = `${activeCount} Active Filter${activeCount > 1 ? 's' : ''}`;
        counterElement.className = 'badge-counter';
    }
}

// Initialize filter count on page load
document.addEventListener('DOMContentLoaded', function() {
    updateFilterCount();
    
    // Update count when form inputs change
    const form = document.getElementById('searchForm');
    const inputs = form.querySelectorAll('input, select');
    inputs.forEach(input => {
        input.addEventListener('change', updateFilterCount);
        input.addEventListener('input', updateFilterCount);
    });
});
</script>
