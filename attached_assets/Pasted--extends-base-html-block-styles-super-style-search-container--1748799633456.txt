<script>
    function updateFilterCount() {
        const form = document.getElementById('searchForm');
        const inputs = form.querySelectorAll('input[type="text"], input[type="date"], input[type="number"], select');
        let activeCount = 0;

        inputs.forEach(input => {
            if (input.value && input.value.trim() !== '') {
                activeCount++;
            }
        });

        const counter = document.getElementById('activeFiltersCount');
        counter.textContent = activeCount === 0
            ? '0 Active Filters'
            : `${activeCount} Active Filter${activeCount > 1 ? 's' : ''}`;
    }

    function exportResults() {
        const params = new URLSearchParams(window.location.search);
        params.set('export', 'csv');
        window.location.href = '{{ url_for("booking.search") }}?' + params.toString();
    }

    document.addEventListener('DOMContentLoaded', () => {
        updateFilterCount();
        document.querySelectorAll('#searchForm input, #searchForm select').forEach(el => {
            el.addEventListener('input', updateFilterCount);
            el.addEventListener('change', updateFilterCount);
        });
    });
</script>
